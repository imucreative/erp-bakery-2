# AUTH MODULE

```prisma
model Role {
  id          String   @id @default(uuid())
  code        String   @unique
  name        String

  users       User[]
  permissions RolePermission[]

  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt
}

model Permission {
  id          String   @id @default(uuid())
  code        String   @unique
  name        String

  roles       RolePermission[]

  createdAt   DateTime @default(now())
}

model RolePermission {
  roleId        String
  permissionId  String

  role          Role       @relation(fields: [roleId], references: [id])
  permission    Permission @relation(fields: [permissionId], references: [id])

  @@id([roleId, permissionId])
}
```

# MASTER MODULE

```prisma
model Outlet {
  id            String      @id @default(uuid())
  code          String      @unique
  name          String
  address       String?
  phone         String?
  isActive      Boolean     @default(true)

  warehouses    Warehouse[]
  users         User[]
  sales         Sale[]

  createdAt     DateTime    @default(now())
  updatedAt     DateTime    @updatedAt
}

model Warehouse {
  id                    String              @id @default(uuid())
  outletId              String

  code                  String              @unique
  name                  String

  type                  WarehouseType
  isProductionSite      Boolean             @default(false)
  isActive              Boolean             @default(true)

  outlet                Outlet             @relation(fields: [outletId], references: [id])

  inventoryBatches      InventoryBatch[]
  productionOrders      ProductionOrder[]
  stockOpnames          StockOpname[]

  createdAt             DateTime           @default(now())
  updatedAt             DateTime           @updatedAt

  @@index([outletId])
}

model User {
  id            String      @id @default(uuid())

  roleId        String
  outletId      String?

  name          String
  email         String      @unique
  password      String
  phone         String?

  isActive      Boolean     @default(true)

  role          Role        @relation(fields: [roleId], references: [id])
  outlet        Outlet?     @relation(fields: [outletId], references: [id])

  createdAt     DateTime    @default(now())
  updatedAt     DateTime    @updatedAt

  @@index([roleId])
}
```

# PRODUCT MODULE

```prisma
model ProductGroup {
  id            String            @id @default(uuid())

  isActive      Boolean           @default(true)

  categories    ProductCategory[]

  id             String       @id @default(uuid())
  productGroupId String
  code           String       @unique
  name           String
  description    String?
  isActive       Boolean      @default(true)
  deletedAt      DateTime?

  productGroup   ProductGroup @relation(fields: [productGroupId], references: [id])

  id            String      @id @default(uuid())

  productGroupId    String
  categoryId        String
  uomId             String
```

# INVENTORY MODULE

```prisma
model InventoryBatch {
  id                String         @id @default(uuid())

  warehouseId       String
  productId         String

  batchNumber       String
  expiredDate       DateTime

  qtyAvailable      Decimal        @db.Decimal(18,2)

  status            BatchStatus    @default(ACTIVE)

  warehouse         Warehouse      @relation(fields: [warehouseId], references: [id])
  product           Product        @relation(fields: [productId], references: [id])

  movements         InventoryMovement[]

  createdAt         DateTime       @default(now())

  @@index([warehouseId])
  @@index([productId])
  @@index([expiredDate])

  @@unique([warehouseId, productId, batchNumber])
}

model InventoryMovement {
  id                String          @id @default(uuid())

  warehouseId       String
  batchId           String
  productId         String

  movementType      MovementType

  qty               Decimal         @db.Decimal(18,2)
  beforeQty         Decimal         @db.Decimal(18,2)
  afterQty          Decimal         @db.Decimal(18,2)

  referenceType     String
  referenceId       String

  notes             String?

  userId            String?

  warehouse         Warehouse       @relation(fields: [warehouseId], references: [id])
  batch             InventoryBatch  @relation(fields: [batchId], references: [id])
  product           Product         @relation(fields: [productId], references: [id])

  createdAt         DateTime        @default(now())

  @@index([warehouseId])
  @@index([productId])
  @@index([referenceType, referenceId])
}

model StockOpname {
  id                String           @id @default(uuid())

  warehouseId       String

  opnameDate        DateTime
  status            String

  notes             String?

  warehouse         Warehouse        @relation(fields: [warehouseId], references: [id])

  items             StockOpnameItem[]

  createdAt         DateTime         @default(now())
}

model StockOpnameItem {
  id                String           @id @default(uuid())

  stockOpnameId     String
  batchId           String

  systemQty         Decimal          @db.Decimal(18,2)
  actualQty         Decimal          @db.Decimal(18,2)
  differenceQty     Decimal          @db.Decimal(18,2)

  stockOpname       StockOpname      @relation(fields: [stockOpnameId], references: [id])

  createdAt         DateTime         @default(now())
}
```

# PRODUCTION MODULE

```prisma
model Recipe {
  id                String           @id @default(uuid())

  productId         String

  version           Int
  yieldQty          Decimal          @db.Decimal(18,2)

  isActive          Boolean          @default(true)

  product           Product          @relation(fields: [productId], references: [id])

  items             RecipeItem[]

  createdAt         DateTime         @default(now())

  @@unique([productId, version])
}

model RecipeItem {
  id                String           @id @default(uuid())

  recipeId          String
  rawMaterialId     String

  qty               Decimal          @db.Decimal(18,2)

  recipe            Recipe           @relation(fields: [recipeId], references: [id])

  createdAt         DateTime         @default(now())
}

model ProductionOrder {
  id                String             @id @default(uuid())

  productionNumber  String             @unique

  warehouseId       String
  recipeId          String

  plannedQty        Decimal            @db.Decimal(18,2)
  actualQty         Decimal            @db.Decimal(18,2)

  productionDate    DateTime

  status            ProductionStatus

  notes             String?

  warehouse         Warehouse          @relation(fields: [warehouseId], references: [id])
  recipe            Recipe             @relation(fields: [recipeId], references: [id])

  materials         ProductionMaterial[]
  results           ProductionResult[]
  wastes            ProductionWaste[]

  createdAt         DateTime           @default(now())
}

model ProductionMaterial {
  id                   String          @id @default(uuid())

  productionOrderId    String
  batchId              String

  qtyConsumed          Decimal         @db.Decimal(18,2)

  productionOrder      ProductionOrder @relation(fields: [productionOrderId], references: [id])

  createdAt            DateTime        @default(now())
}

model ProductionResult {
  id                   String          @id @default(uuid())

  productionOrderId    String

  batchNumber          String
  expiredDate          DateTime

  qtyProduced          Decimal         @db.Decimal(18,2)

  productionOrder      ProductionOrder @relation(fields: [productionOrderId], references: [id])
}

model ProductionWaste {
  id                   String          @id @default(uuid())

  productionOrderId    String
  productId            String

  qty                  Decimal         @db.Decimal(18,2)

  notes                String?

  productionOrder      ProductionOrder @relation(fields: [productionOrderId], references: [id])
}
```

# SALES MODULE

```prisma
model Customer {
  id            String      @id @default(uuid())

  name          String
  phone         String?
  email         String?
  address       String?

  sales         Sale[]
}

model Shift {
  id              String      @id @default(uuid())

  outletId        String
  cashierId       String

  openAt          DateTime
  closeAt         DateTime?

  openingCash     Decimal     @db.Decimal(18,2)
  closingCash     Decimal?    @db.Decimal(18,2)

  status          String

  sales           Sale[]

  createdAt       DateTime    @default(now())
}

model Sale {
  id                String            @id @default(uuid())

  invoiceNumber     String            @unique

  outletId          String
  shiftId           String

  customerId        String?

  subtotal          Decimal           @db.Decimal(18,2)
  discount          Decimal           @db.Decimal(18,2)
  tax               Decimal           @db.Decimal(18,2)

  total             Decimal           @db.Decimal(18,2)

  paymentStatus     PaymentStatus

  salesChannel      SalesChannel

  status            String

  customer          Customer?         @relation(fields: [customerId], references: [id])

  shift             Shift             @relation(fields: [shiftId], references: [id])

  items             SaleItem[]
  payments          Payment[]

  createdAt         DateTime          @default(now())
}

model SaleItem {
  id                String            @id @default(uuid())

  saleId            String
  batchId           String
  productId         String

  qty               Decimal           @db.Decimal(18,2)
  price             Decimal           @db.Decimal(18,2)
  subtotal          Decimal           @db.Decimal(18,2)

  sale              Sale              @relation(fields: [saleId], references: [id])
}

model Payment {
  id                String            @id @default(uuid())

  saleId            String

  paymentMethod     String

  amount            Decimal           @db.Decimal(18,2)

  gatewayReference  String?

  status            String

  paidAt            DateTime?

  sale              Sale              @relation(fields: [saleId], references: [id])

  createdAt         DateTime          @default(now())
}
```

# Notes

1. Gunakan Soft Delete pada seluruh master table menggunakan field:

```prisma
deletedAt DateTime?
```

2. Semua proses berikut wajib menggunakan Prisma Transaction:

* Sales
* Production
* Transfer
* Stock Adjustment
* Replenishment Fulfillment

3. Implementasikan FEFO di Service Layer, bukan di Prisma Model.
